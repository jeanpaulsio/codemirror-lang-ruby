# Variable assignment

x = 42

==>

Program(Assignment(Identifier, AssignOp, Integer))

# Instance variable assignment

@name = "Alice"

==>

Program(Assignment(InstanceVariable, AssignOp, InterpolatedString(StringContent)))

# Method call

puts("hello")

==>

Program(ExpressionStatement(MethodCall(Identifier, ArgList(InterpolatedString(StringContent)))))

# Chained method call

foo.bar

==>

Program(ExpressionStatement(MethodCall(Identifier, Identifier)))

# Block with braces

f = -> (x) { x + 1 }

==>

Program(Assignment(Identifier, AssignOp, Lambda(ParamList(Param(Identifier)), Block(ExpressionStatement(BinaryExpression(Identifier, ArithOp, Integer))))))

# Do block

f = -> (x) do
  puts(x)
end

==>

Program(Assignment(Identifier, AssignOp, Lambda(ParamList(Param(Identifier)), DoBlock(do, ExpressionStatement(MethodCall(Identifier, ArgList(Identifier))), end))))

# Lambda

f = -> (x) { x }

==>

Program(Assignment(Identifier, AssignOp, Lambda(ParamList(Param(Identifier)), Block(ExpressionStatement(Identifier)))))
